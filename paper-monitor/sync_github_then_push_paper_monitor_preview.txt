schtasks /Create /TN "Paper Monitor Git Sync 20_30" /SC DAILY /ST 20:30 /TR "cmd.exe /c \"\"E:\MY FILES\Onedrive\CV\dm-dut.github.io\paper-monitor\sync_github_then_push_paper_monitor.bat\"\"" /F



@echo off
chcp 65001 >nul
setlocal EnableExtensions EnableDelayedExpansion

REM ============================================================
REM GitHub root pull -> Paper Monitor push
REM Place this BAT in: dm-dut.github.io\paper-monitor\
REM
REM Guaranteed order:
REM   FIRST PULL succeeds
REM   -> stage/commit paper-monitor only
REM   -> SECOND PULL succeeds
REM   -> push
REM
REM Any push retry is also preceded by a successful pull/rebase.
REM ============================================================

set "PROJECT_ROOT=%~dp0"
for %%I in ("%PROJECT_ROOT%..") do set "REPO_ROOT=%%~fI"

set "REMOTE=origin"
set "BRANCH=main"
set "LOCK_DIR=%TEMP%\paper_monitor_git_sync_2030.lock"
set "LOG_DIR=%LOCALAPPDATA%\PaperMonitor"
set "LOG_FILE=%LOG_DIR%\git_sync_2030.log"
set "EXIT_CODE=0"

if not exist "%LOG_DIR%" mkdir "%LOG_DIR%" >nul 2>&1

2>nul mkdir "%LOCK_DIR%"
if errorlevel 1 (
    echo Another sync instance appears to be running.
    exit /b 2
)

echo.>> "%LOG_FILE%"
echo ============================================================>> "%LOG_FILE%"
echo Git sync started: %date% %time%>> "%LOG_FILE%"
echo Repository: %REPO_ROOT%>> "%LOG_FILE%"

echo ============================================================
echo GitHub Root Pull -^> Paper Monitor Push
echo ============================================================

REM 1. Verify repository/branch/remote.
echo.
echo [1/7] Verifying repository...

git -C "%REPO_ROOT%" rev-parse --is-inside-work-tree >nul 2>&1
if errorlevel 1 (
    echo ERROR: Repository root is invalid.
    set "EXIT_CODE=1"
    goto CLEANUP
)

set "CURRENT_BRANCH="
for /f "delims=" %%B in ('git -C "%REPO_ROOT%" branch --show-current 2^>nul') do set "CURRENT_BRANCH=%%B"

if not defined CURRENT_BRANCH (
    echo ERROR: Could not determine current branch.
    set "EXIT_CODE=1"
    goto CLEANUP
)

if /I not "%CURRENT_BRANCH%"=="%BRANCH%" (
    echo ERROR: Current branch is "%CURRENT_BRANCH%", expected "%BRANCH%".
    set "EXIT_CODE=1"
    goto CLEANUP
)

git -C "%REPO_ROOT%" remote get-url "%REMOTE%" >nul 2>&1
if errorlevel 1 (
    echo ERROR: Remote "%REMOTE%" does not exist.
    set "EXIT_CODE=1"
    goto CLEANUP
)

REM 2. FIRST PULL: if this fails, absolutely no push.
echo.
echo [2/7] FIRST PULL from GitHub repository root...
echo FIRST PULL started.>> "%LOG_FILE%"

git -C "%REPO_ROOT%" pull --rebase --autostash "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if errorlevel 1 (
    echo ERROR: FIRST PULL failed.
    echo IMPORTANT: Push is blocked.
    echo FIRST PULL FAILED - PUSH BLOCKED.>> "%LOG_FILE%"
    set "EXIT_CODE=1"
    goto CLEANUP
)

echo FIRST PULL completed successfully.
echo FIRST PULL completed successfully.>> "%LOG_FILE%"

REM 3. Safety: git push is repository-level, so ensure any existing
REM unpushed commits affect paper-monitor/ only.
echo.
echo [3/7] Checking existing unpushed commits...

call :CHECK_AHEAD_SCOPE
if errorlevel 1 (
    echo ERROR: Existing unpushed commits contain files outside paper-monitor/.
    echo Push blocked to avoid sending unrelated repository changes.
    set "EXIT_CODE=1"
    goto CLEANUP
)

REM 4. Stage/commit only paper-monitor/.
echo.
echo [4/7] Staging Paper Monitor folder only...

git -C "%REPO_ROOT%" add -A -- "paper-monitor/"
if errorlevel 1 (
    echo ERROR: git add paper-monitor/ failed.
    set "EXIT_CODE=1"
    goto CLEANUP
)

git -C "%REPO_ROOT%" diff --cached --quiet -- "paper-monitor/"
if errorlevel 1 (
    echo Paper Monitor changes detected. Creating commit...
    git -C "%REPO_ROOT%" commit -m "Update Paper Monitor %date% %time%" -- "paper-monitor/" >> "%LOG_FILE%" 2>&1
    if errorlevel 1 (
        echo ERROR: Paper Monitor commit failed.
        set "EXIT_CODE=1"
        goto CLEANUP
    )
) else (
    echo No new Paper Monitor changes to commit.
)

REM 5. SECOND PULL immediately before push.
echo.
echo [5/7] SECOND PULL immediately before push...
echo SECOND PULL started.>> "%LOG_FILE%"

git -C "%REPO_ROOT%" pull --rebase --autostash "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if errorlevel 1 (
    echo ERROR: SECOND PULL failed.
    echo IMPORTANT: Push is blocked.
    echo SECOND PULL FAILED - PUSH BLOCKED.>> "%LOG_FILE%"
    set "EXIT_CODE=1"
    goto CLEANUP
)

echo SECOND PULL completed successfully.
echo SECOND PULL completed successfully.>> "%LOG_FILE%"

call :CHECK_AHEAD_SCOPE
if errorlevel 1 (
    echo ERROR: Post-pull scope check failed.
    echo Push blocked.
    set "EXIT_CODE=1"
    goto CLEANUP
)

REM 6. Push attempt 1, only after successful SECOND PULL.
echo.
echo [6/7] Pushing after successful pull...

git -C "%REPO_ROOT%" push "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if not errorlevel 1 goto PUSH_SUCCESS

REM 7. Retry only after another successful pull.
echo.
echo Push attempt 1 failed.
echo Waiting 10 seconds, then pulling again BEFORE retry...
timeout /t 10 /nobreak >nul

echo.
echo [7/7] Recovery pull before push retry...

git -C "%REPO_ROOT%" pull --rebase --autostash "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if errorlevel 1 (
    echo ERROR: Recovery pull failed. Retry push blocked.
    set "EXIT_CODE=1"
    goto CLEANUP
)

call :CHECK_AHEAD_SCOPE
if errorlevel 1 (
    echo ERROR: Recovery scope check failed. Retry push blocked.
    set "EXIT_CODE=1"
    goto CLEANUP
)

git -C "%REPO_ROOT%" push "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if not errorlevel 1 goto PUSH_SUCCESS

REM Final retry: again, pull first.
echo Push attempt 2 failed.
echo Waiting 15 seconds, then doing FINAL PULL...
timeout /t 15 /nobreak >nul

git -C "%REPO_ROOT%" pull --rebase --autostash "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if errorlevel 1 (
    echo ERROR: Final pull failed. Final push blocked.
    set "EXIT_CODE=1"
    goto CLEANUP
)

call :CHECK_AHEAD_SCOPE
if errorlevel 1 (
    echo ERROR: Final scope check failed. Final push blocked.
    set "EXIT_CODE=1"
    goto CLEANUP
)

git -C "%REPO_ROOT%" push "%REMOTE%" "%BRANCH%" >> "%LOG_FILE%" 2>&1
if errorlevel 1 (
    echo ERROR: Push failed after all attempts.
    echo Check log: %LOG_FILE%
    set "EXIT_CODE=1"
    goto CLEANUP
)

:PUSH_SUCCESS
echo.
echo ============================================================
echo SUCCESS
echo GitHub root was pulled successfully BEFORE push.
echo Paper Monitor changes were pushed to %REMOTE%/%BRANCH%.
echo ============================================================
echo SUCCESS: %date% %time%>> "%LOG_FILE%"
set "EXIT_CODE=0"
goto CLEANUP

:CHECK_AHEAD_SCOPE
set "OUTSIDE_AHEAD=0"
set "AHEAD_COUNT=0"

for /f "delims=" %%N in ('git -C "%REPO_ROOT%" rev-list --count "%REMOTE%/%BRANCH%..HEAD" 2^>nul') do set "AHEAD_COUNT=%%N"

if "!AHEAD_COUNT!"=="0" (
    echo No existing unpushed commits.
    exit /b 0
)

echo Local branch is ahead by !AHEAD_COUNT! commit(s).

for /f "usebackq delims=" %%F in (`git -C "%REPO_ROOT%" diff --name-only "%REMOTE%/%BRANCH%..HEAD" 2^>nul`) do (
    set "CHECK_PATH=%%F"
    if /I not "!CHECK_PATH:~0,14!"=="paper-monitor/" (
        echo OUTSIDE PAPER MONITOR: !CHECK_PATH!
        echo OUTSIDE PAPER MONITOR: !CHECK_PATH!>> "%LOG_FILE%"
        set "OUTSIDE_AHEAD=1"
    )
)

if "!OUTSIDE_AHEAD!"=="1" exit /b 1

echo All unpushed commit changes are inside paper-monitor/.
exit /b 0

:CLEANUP
if exist "%LOCK_DIR%" rmdir "%LOCK_DIR%" >nul 2>&1
echo Git sync ended: %date% %time% with code %EXIT_CODE%>> "%LOG_FILE%"
exit /b %EXIT_CODE%
